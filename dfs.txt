#include <iostream>
#include <vector>
#include <omp.h>

using namespace std;

vector<vector<int>> graph;
vector<int> visited;

void dfs(int node) {

    visited[node] = 1;

    cout << node << " ";

    #pragma omp parallel for
    for (int i = 0; i < graph[node].size(); i++) {

        int next = graph[node][i];

        if (!visited[next]) {

            #pragma omp task
            {
                if (!visited[next]) {
                    dfs(next);
                }
            }
        }
    }
}

int main() {

    int n, e;
    cin >> n >> e;

    graph.resize(n);
    visited.resize(n, 0);

    for (int i = 0; i < e; i++) {

        int u, v;
        cin >> u >> v;

        graph[u].push_back(v);
        graph[v].push_back(u);
    }

    cout << "Parallel DFS: ";

    dfs(0);

    return 0;
}
