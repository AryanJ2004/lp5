#include <iostream>
#include <vector>
#include <omp.h>

using namespace std;

// Merge Function
void merge(vector<int> &arr, int st, int mid, int end) {

    vector<int> temp;

    int i = st;
    int j = mid + 1;
    
    while(i <= mid && j <= end) {

        if(arr[i] <= arr[j]) {
            temp.push_back(arr[i]);
            i++;
        }
        else {
            temp.push_back(arr[j]);
            j++;
        }
    }

    // Remaining elements of left half
    while(i <= mid) {
        temp.push_back(arr[i]);
        i++;
    }

    // Remaining elements of right half
    while(j <= end) {
        temp.push_back(arr[j]);
        j++;
    }

    // Copy merged elements back to original array
    for(int idx = 0; idx < temp.size(); idx++) {
        arr[st + idx] = temp[idx];
    }
}


// Sequential Merge Sort
void seqMergeSort(vector<int> &arr, int st, int end) {

    if(st >= end)
        return;

    int mid = (st + end) / 2;

    seqMergeSort(arr, st, mid);
    seqMergeSort(arr, mid + 1, end);

    merge(arr, st, mid, end);
}


// Parallel Merge Sort
void parMergeSort(vector<int> &arr, int st, int end) {

    if(st >= end)
        return;

    int mid = (st + end) / 2;

    #pragma omp parallel sections
    {
        #pragma omp section
        parMergeSort(arr, st, mid);

        #pragma omp section
        parMergeSort(arr, mid + 1, end);
    }

    merge(arr, st, mid, end);
}


int main() {

   int n = 1000;

vector<int> a(n);

for(int i = 0; i < n; i++) {
    a[i] = rand();
}

vector<int> b = a;

    double start, end;

    // Sequential Merge Sort
    start = omp_get_wtime();

    seqMergeSort(a, 0, a.size() - 1);

    end = omp_get_wtime();

    cout << "Sequential Merge Sort Time: "
         << end - start << " sec\n";

    cout << "Sequential Sorted Array: ";

    for(int x : a)
        cout << x << " ";

    cout << "\n\n";


    // Parallel Merge Sort
    start = omp_get_wtime();

    parMergeSort(b, 0, b.size() - 1);

    end = omp_get_wtime();

    cout << "Parallel Merge Sort Time: "
         << end - start << " sec\n";

    cout << "Parallel Sorted Array: ";

    for(int x : b)
        cout << x << " ";

    cout << endl;

    return 0;
}
