#include <bits/stdc++.h>
#include <omp.h>
using namespace std;

int main()
{
    vector<int>a(100);
    for(int i=0;i<100;i++){
        a[i]=rand();
    }
    int n = 100;

    // Sequential Bubble Sort
    double start = omp_get_wtime();

    for (int i = 0; i < n - 1; i++)
    {
        for (int j = 0; j < n - i - 1; j++)
        {
            if (a[j] > a[j + 1])
            {
                swap(a[j], a[j + 1]);
            }
        }
    }

    double end = omp_get_wtime();

    cout << "Sequential Time: " << end - start << endl;

    cout << "Sorted Array: ";
    for (int i = 0; i < n; i++)
    {
        cout << a[i] << " ";
    }

    cout << endl;

    // Parallel Bubble Sort
  vector<int>b(100);
    for(int i=0;i<100;i++){
        b[i]=rand();
    }

    start = omp_get_wtime();

    for (int i = 0; i < n; i++)
    {
        // Even phase
        if (i % 2 == 0)
        {
            #pragma omp parallel for
            for (int j = 0; j < n - 1; j += 2)
            {
                if (b[j] > b[j + 1])
                {
                    swap(b[j], b[j + 1]);
                }
            }
        }

        // Odd phase
        else
        {
            #pragma omp parallel for
            for (int j = 1; j < n - 1; j += 2)
            {
                if (b[j] > b[j + 1])
                {
                    swap(b[j], b[j + 1]);
                }
            }
        }
    }

    end = omp_get_wtime();

    cout << "Parallel Time: " << end - start << endl;

    cout << "Sorted Array: ";
    for (int i = 0; i < n; i++)
    {
        cout << b[i] << " ";
    }

    return 0;
}
