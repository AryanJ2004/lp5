#include <iostream>
#include <omp.h>
using namespace std;

int main()
{

    int a[] = {5, 1, 4, 2, 8};
    int n = 5;

    // Sequential Bubble Sort
    double start = omp_get_wtime();

    for (int i = 0; i < n - 1; i++)
    {
        for (int j = 0; j < n - i - 1; j++)
        {
            if (a[j] > a[j + 1])
            {
                swap(a[j], a[j + 1]);
            }
        }
    }

    double end = omp_get_wtime();

    cout << "Sequential Time: " << end - start << endl;

    cout << "Sorted Array: ";
    for (int i = 0; i < n; i++)
    {
        cout << a[i] << " ";
    }

    cout << endl;

    // Parallel Bubble Sort
    int b[] = {5, 1, 4, 2, 8};

    start = omp_get_wtime();

    for (int i = 0; i < n - 1; i++)
    {

#pragma omp parallel for
        for (int j = 0; j < n - i - 1; j++)
        {

            if (b[j] > b[j + 1])
            {
                swap(b[j], b[j + 1]);
            }
        }
    }

    end = omp_get_wtime();

    cout << "Parallel Time: " << end - start << endl;

    cout << "Sorted Array: ";
    for (int i = 0; i < n; i++)
    {
        cout << b[i] << " ";
    }

    return 0;
}
